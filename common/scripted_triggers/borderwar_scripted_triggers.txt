# ROOT と FROM が同じイデオロギー（民主主義・ファシズム・共産主義・中道）か
bw_same_ideology_as_from = {
	OR = {
		AND = {
			has_government = democratic
			FROM = { has_government = democratic }
		}
		AND = {
			has_government = fascism
			FROM = { has_government = fascism }
		}
		AND = {
			has_government = communism
			FROM = { has_government = communism }
		}
		AND = {
			has_government = neutrality
			FROM = { has_government = neutrality }
		}
	}
}

# 要求した国（FROM）が自国（ROOT）の属国である＝自国が宗主国
bw_from_is_subject = {
	FROM = { is_subject_of = ROOT }
}

# 要求した国（FROM）が同盟国（同じ陣営）か宗主国で、割譲を断れない
# 自国が FROM の宗主国である場合は除く
bw_must_cede_to_from = {
	NOT = { bw_from_is_subject = yes }
	OR = {
		is_in_faction_with = FROM
		is_subject_of = FROM
	}
}
